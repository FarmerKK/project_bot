⏰ เครื่องมือสุ่มวันเวลาใน Google Sheet

เนื่องจาก Google Form เซิร์ฟเวอร์จะประทับเวลา ณ วินาทีที่ได้รับข้อมูลเสมอ หากต้องการให้วันเวลากระจายตัวย้อนหลังอย่างสมจริง ให้ใช้สคริปต์นี้ใน Google Sheet:

  1. เปิด Google Sheet คำตอบของฟอร์ม
  2. ไปที่เมนู ส่วนขยาย (Extensions) → Apps Script
  3. วางโค้ดด้านล่างแล้วกดปุ่ม บันทึก (Save) และกด เรียกใช้ (Run)
function randomizeTimestamps() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var lastRow = sheet.getLastRow(); if (lastRow < 2) return; var startDate = new Date(2026, 8, 1); // 1 ก.ย. var endDate = new Date(2026, 8, 12); // 12 ก.ย. var dates = []; for (var i = 2; i <= lastRow; i++) { var randomTime = startDate.getTime() + Math.random() * (endDate.getTime() - startDate.getTime()); var d = new Date(randomTime); d.setHours(8 + Math.floor(Math.random() * 14)); d.setMinutes(Math.floor(Math.random() * 60)); d.setSeconds(Math.floor(Math.random() * 60)); dates.push(d); } dates.sort(function(a, b) { return a - b; }); var output = dates.map(function(d) { return [d]; }); sheet.getRange(2, 1, output.length, 1).setValues(output); SpreadsheetApp.getUi().alert("✔ สุ่มกระจายวันเวลาเรียบร้อยแล้วทั้งหมด " + output.length + " แถว!"); }
